{{- /* A page's cover is the photo named cover.* in its own folder. An entry with no cover falls back to its first photo by name, so a log line gets a thumbnail without anyone having to nominate one. Returns nothing when the page has no photos at all - every caller has to cope with that. */ -}} {{- $cover := false -}} {{- with .Resources.GetMatch "cover.*" -}} {{- $cover = . -}} {{- else -}} {{- if .IsPage -}} {{- with index (.Resources.ByType "image") 0 -}}{{- $cover = . -}}{{- end -}} {{- end -}} {{- end -}} {{- return $cover -}}

{{- /* The top of every page in a project: the project's name and a row of tabs - Overview, Log, Photos, Reference - so the reader always knows which project they are in and can get to any part of it in one step. Takes dict "page" "current" "full" . On the project's own page ("full") the name is the page's headline, under its byline; everywhere else it is a link back, kept small so the page's own title leads. The project's summary is not shown: the reader has just read it on the home page, and it stays in the description that link previews use. A tab with nothing in it is left out. */ -}} {{- $page := .page }} {{- $project := partial "project-of.html" $page }} {{- with $page.Parent }}{{ if eq .Path "/projects" }}{{ $project = $page }}{{ end }}{{ end }} {{- $current := .current }} {{- $log := partial "log.html" $project }} {{- $photos := partial "project-photos.html" $project }} {{- $refs := where $project.RegularPagesRecursive "Path" "like" `/reference/` }} {{- $tabs := slice (dict "name" "overview" "label" "Overview" "page" $project "count" 0) }} {{- with $project.GetPage "log" }}{{ if $log }}{{ $tabs = $tabs | append (dict "name" "log" "label" "Log" "page" . "count" (len $log)) }}{{ end }}{{ end }} {{- with $project.GetPage "photos" }}{{ if $photos }}{{ $tabs = $tabs | append (dict "name" "photos" "label" "Photos" "page" . "count" (len $photos)) }}{{ end }}{{ end }} {{- with $project.GetPage "reference" }}{{ if $refs }}{{ $tabs = $tabs | append (dict "name" "reference" "label" "Reference" "page" . "count" (len $refs)) }}{{ end }}{{ end }}
